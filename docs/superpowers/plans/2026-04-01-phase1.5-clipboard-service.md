**Goal:** Build a standalone FastAPI service (`:8766`) that deterministically generates and validates C3 clipboard JSON from Intent IR, with zero LLM dependency.

**Architecture:** Schema-driven generator + structural validator. The service receives precise Intent IR from Copilot Core, maps it to clipboard JSON via code templates, and validates the output against format schemas. ACE ID validation delegates to RAG `:8765` via HTTP; if RAG is unavailable, ACE checks are skipped with warnings. An error notebook (JSONL) captures failures for regression testing.

- [ ] **Step 4: Implement StructuralValidator**

Create `src/validator/structural.py`. This is the largest single file — extracted from legacy `validate_output.py` and `clipboard.py:ClipboardValidator`, refactored to use a `ValidationResult` dataclass instead of returning tuples.

```python
"""
Structural validation for Construct 3 clipboard JSON.

Validates format rules without requiring ACE schema data:
- Clipboard header and type
- Event structure (blocks, variables, functions, groups, comments)
- Object type structure (plugin-id, animations, behaviors)
- Layout structure (layers, instances)
- Parameter format rules (comparison operators, empty params)
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field


@dataclass
class ValidationResult:
    passed: bool = True
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)


VALID_CLIPBOARD_TYPES = {
    "events", "conditions", "actions",
    "object-types", "world-instances",
    "layouts", "event-sheets",
}

VALID_EVENT_TYPES = {"comment", "variable", "group", "block", "function-block"}
VALID_VAR_TYPES = {"number", "string", "boolean"}
VALID_FUNCTION_RETURN_TYPES = {"none", "number", "string", "any"}
COMPARISON_OPERATORS = {0, 1, 2, 3, 4, 5}


class StructuralValidator:
    """Validate C3 clipboard JSON structural rules (no ACE lookup)."""

    def validate(self, data: dict) -> ValidationResult:
        self._result = ValidationResult()

        if not isinstance(data, dict):
            self._error("Root must be a JSON object")
            return self._finish()

        if not data.get("is-c3-clipboard-data"):
            self._error("Missing 'is-c3-clipboard-data': true")

        clip_type = data.get("type")
        if clip_type not in VALID_CLIPBOARD_TYPES:
            self._error(f"Invalid type: '{clip_type}', must be one of {sorted(VALID_CLIPBOARD_TYPES)}")
            return self._finish()

        items = data.get("items")
        if not isinstance(items, list):
            self._error("'items' must be an array")
            return self._finish()

        dispatch = {
            "events": self._validate_events,
            "conditions": self._validate_conditions_list,
            "actions": self._validate_actions_list,
            "object-types": self._validate_object_types,
            "world-instances": self._validate_world_instances,
            "layouts": self._validate_layouts,
            "event-sheets": self._validate_event_sheets,
        }
        handler = dispatch.get(clip_type)
        if handler:
            handler(data)

        return self._finish()

    # -- Events ----------------------------------------------------------

    def _validate_events(self, data: dict):
        for i, item in enumerate(data.get("items", [])):
            self._validate_event(item, f"items[{i}]")

    def _validate_event(self, event: dict, path: str):
        if not isinstance(event, dict):
            self._error(f"{path}: must be an object")
            return
        event_type = event.get("eventType")
        if event_type not in VALID_EVENT_TYPES:
            self._error(f"{path}: invalid eventType '{event_type}'")
            return
        if event_type == "comment":
            if "text" not in event:
                self._error(f"{path}: comment missing 'text'")
        elif event_type == "variable":
            self._validate_variable(event, path)
        elif event_type == "group":
            self._validate_group(event, path)
        elif event_type == "block":
            self._validate_block(event, path)
        elif event_type == "function-block":
            self._validate_function_block(event, path)

    def _validate_variable(self, var: dict, path: str):
        if "name" not in var:
            self._error(f"{path}: variable missing 'name'")
        if var.get("type") not in VALID_VAR_TYPES:
            self._error(f"{path}: invalid variable type '{var.get('type')}'")
        if "initialValue" not in var:
            self._warn(f"{path}: variable should set 'initialValue'")
        if "comment" not in var:
            self._warn(f"{path}: variable missing 'comment' field (can be empty string)")

    def _validate_group(self, group: dict, path: str):
        if "title" not in group:
            self._error(f"{path}: group missing 'title'")
        for i, child in enumerate(group.get("children", [])):
            self._validate_event(child, f"{path}.children[{i}]")

    def _validate_block(self, block: dict, path: str):
        if "conditions" not in block:
            self._error(f"{path}: block missing 'conditions' array")
        elif isinstance(block["conditions"], list):
            for i, c in enumerate(block["conditions"]):
                self._validate_condition(c, f"{path}.conditions[{i}]")

        if "actions" not in block:
            self._error(f"{path}: block missing 'actions' array")
        elif isinstance(block["actions"], list):
            for i, a in enumerate(block["actions"]):
                self._validate_action(a, f"{path}.actions[{i}]")

        for i, child in enumerate(block.get("children", [])):
            self._validate_event(child, f"{path}.children[{i}]")

    def _validate_function_block(self, func: dict, path: str):
        if "functionName" not in func:
            self._error(f"{path}: function-block missing 'functionName'")
        ret = func.get("functionReturnType", "none")
        if ret not in VALID_FUNCTION_RETURN_TYPES:
            self._error(f"{path}: invalid return type '{ret}'")
        self._validate_block(func, path)

    # -- Conditions / Actions -------------------------------------------

    def _validate_conditions_list(self, data: dict):
        for i, c in enumerate(data.get("items", [])):
            self._validate_condition(c, f"items[{i}]")

    def _validate_condition(self, cond: dict, path: str):
        if not isinstance(cond, dict):
            self._error(f"{path}: condition must be an object")
            return
        if "callFunction" in cond:
            return
        if "id" not in cond:
            self._error(f"{path}: condition missing 'id'")
        if "objectClass" not in cond:
            self._error(f"{path}: condition missing 'objectClass'")
        self._check_parameters(cond, path)

    def _validate_actions_list(self, data: dict):
        for i, a in enumerate(data.get("items", [])):
            self._validate_action(a, f"items[{i}]")

    def _validate_action(self, action: dict, path: str):
        if not isinstance(action, dict):
            self._error(f"{path}: action must be an object")
            return
        # Script action
        if action.get("type") == "script":
            self._validate_script_action(action, path)
            return
        # Comment action
        if action.get("type") == "comment":
            if "text" not in action:
                self._error(f"{path}: action comment missing 'text'")
            return
        # Function call
        if "callFunction" in action:
            return
        if "id" not in action:
            self._error(f"{path}: action missing 'id'")
        if "objectClass" not in action:
            self._error(f"{path}: action missing 'objectClass'")
        self._check_parameters(action, path)

    def _validate_script_action(self, action: dict, path: str):
        lang = action.get("language")
        if lang not in ("javascript", "typescript"):
            self._error(f"{path}: script action invalid 'language': '{lang}'")
        script = action.get("script")
        if not isinstance(script, list) or not all(isinstance(s, str) for s in script):
            self._error(f"{path}: script 'script' must be an array of strings")

    def _check_parameters(self, node: dict, path: str):
        params = node.get("parameters")
        if params is None:
            return  # omitted is fine
        if not isinstance(params, dict):
            self._error(f"{path}: 'parameters' must be an object")
            return
        # Dev notes rule: empty {} should be omitted
        if len(params) == 0:
            self._warn(f"{path}: empty 'parameters' {{}} should be omitted")
        # Check comparison operator range
        comp = params.get("comparison")
        if comp is not None and isinstance(comp, int) and comp not in COMPARISON_OPERATORS:
            self._error(f"{path}.parameters.comparison: invalid operator {comp}, valid: 0-5")

    # -- Object Types ---------------------------------------------------

    def _validate_object_types(self, data: dict):
        for i, item in enumerate(data.get("items", [])):
            self._validate_object_type(item, f"items[{i}]")
        # imageData cross-reference
        image_data = data.get("imageData")
        if isinstance(image_data, list):
            self._validate_imagedata_entries(image_data, "imageData")

    def _validate_object_type(self, obj: dict, path: str):
        if "name" not in obj:
            self._error(f"{path}: missing 'name'")
        if "plugin-id" not in obj:
            self._error(f"{path}: missing 'plugin-id'")
        # Dev notes rule: effectTypes must be array, not object
        et = obj.get("effectTypes")
        if et is not None and not isinstance(et, list):
            self._error(f"{path}: 'effectTypes' must be an array, got {type(et).__name__}")
        # instanceVariables must be array
        iv = obj.get("instanceVariables")
        if iv is not None and not isinstance(iv, list):
            self._error(f"{path}: 'instanceVariables' must be an array")
        # behaviorTypes must be array
        bt = obj.get("behaviorTypes")
        if bt is not None and not isinstance(bt, list):
            self._error(f"{path}: 'behaviorTypes' must be an array")

    # -- World Instances ------------------------------------------------

    def _validate_world_instances(self, data: dict):
        for i, item in enumerate(data.get("items", [])):
            path = f"items[{i}]"
            if "type" not in item:
                self._error(f"{path}: missing 'type'")
            w = item.get("world")
            if not isinstance(w, dict):
                self._error(f"{path}: missing 'world' object")
            elif "x" not in w or "y" not in w:
                self._error(f"{path}: world missing 'x' or 'y'")

    # -- Layouts --------------------------------------------------------

    def _validate_layouts(self, data: dict):
        for i, layout in enumerate(data.get("items", [])):
            path = f"items[{i}]"
            if "name" not in layout:
                self._error(f"{path}: missing 'name'")
            layers = layout.get("layers")
            if not isinstance(layers, list):
                self._error(f"{path}: missing 'layers' array")
            else:
                for j, layer in enumerate(layers):
                    lpath = f"{path}.layers[{j}]"
                    if "name" not in layer:
                        self._error(f"{lpath}: missing 'name'")

    # -- Event Sheets ---------------------------------------------------

    def _validate_event_sheets(self, data: dict):
        for i, sheet in enumerate(data.get("items", [])):
            path = f"items[{i}]"
            if "name" not in sheet:
                self._error(f"{path}: missing 'name'")
            events = sheet.get("events")
            if not isinstance(events, list):
                self._error(f"{path}: missing 'events' array")
            else:
                for j, ev in enumerate(events):
                    self._validate_event(ev, f"{path}.events[{j}]")

    # -- ImageData ------------------------------------------------------

    def _validate_imagedata_entries(self, entries: list, path: str):
        for i, entry in enumerate(entries):
            p = f"{path}[{i}]"
            if not isinstance(entry, str):
                self._error(f"{p}: imageData entry must be a string")
            elif not entry.startswith("data:image/png;base64,"):
                self._error(f"{p}: imageData must be a PNG data URI")

    # -- Helpers --------------------------------------------------------

    def _error(self, msg: str):
```

```python
"""
Low-level node builders for C3 clipboard JSON.

Every builder enforces schema rules by construction:
- No empty parameters dicts (omit instead)
- Variables always have comment field
- Correct eventType values
"""
from __future__ import annotations

from typing import Any


def build_action(
    ace_id: str,
    object_class: str,
    *,
    parameters: dict[str, Any] | None = None,
    behavior_type: str | None = None,
) -> dict:
    node: dict[str, Any] = {"id": ace_id, "objectClass": object_class}
    if behavior_type:
        node["behaviorType"] = behavior_type
    if parameters:  # omit empty
        node["parameters"] = parameters
    return node


def build_condition(
    ace_id: str,
    object_class: str,
    *,
    parameters: dict[str, Any] | None = None,
    behavior_type: str | None = None,
    inverted: bool = False,
) -> dict:
    node: dict[str, Any] = {"id": ace_id, "objectClass": object_class}
    if behavior_type:
        node["behaviorType"] = behavior_type
    if parameters:
        node["parameters"] = parameters
    if inverted:
        node["isInverted"] = True
    return node


def build_variable(
    name: str,
    var_type: str = "number",
    initial_value: str = "0",
    comment: str = "",
) -> dict:
    return {
        "eventType": "variable",
        "name": name,
        "type": var_type,
        "initialValue": initial_value,
        "comment": comment,
    }


def build_comment(text: str) -> dict:
    return {"eventType": "comment", "text": text}


def build_group(
    title: str,
    *,
    children: list[dict] | None = None,
    description: str = "",
    disabled: bool = False,
    is_active_on_start: bool = True,
) -> dict:
    return {
        "eventType": "group",
        "disabled": disabled,
        "title": title,
        "description": description,
        "isActiveOnStart": is_active_on_start,
        "children": children or [],
    }


def build_block(
    conditions: list[dict],
    actions: list[dict],
    *,
    children: list[dict] | None = None,
) -> dict:
    node: dict[str, Any] = {
        "eventType": "block",
        "conditions": conditions,
        "actions": actions,
    }
    if children:
        node["children"] = children
    return node


def build_script_action(
    lines: list[str],
    language: str = "javascript",
) -> dict:
    return {"type": "script", "language": language, "script": lines}


def build_function_block(
    name: str,
    *,
    return_type: str = "none",
    parameters: list[dict] | None = None,
    conditions: list[dict] | None = None,
    actions: list[dict] | None = None,
    description: str = "",
    is_async: bool = False,
) -> dict:
    return {
        "eventType": "function-block",
        "functionName": name,
        "functionDescription": description,
        "functionReturnType": return_type,
        "functionCopyPicked": False,
        "functionIsAsync": is_async,
        "functionParameters": parameters or [],
        "conditions": conditions or [],
        "actions": actions or [],
    }


def wrap_clipboard(
    clip_type: str,
    items: list[dict],
    **extra,
) -> dict:
    data: dict[str, Any] = {
        "is-c3-clipboard-data": True,
        "type": clip_type,
        "items": items,
    }
    data.update(extra)
    return data
```

```python
from typing import Any

from src.generator.builder import wrap_clipboard

}

NO_ANIMATION_PLUGINS = {
    "Text", "SpriteFont", "TiledBg", "NinePatch",
    "Particles", "HTMLElement", "TextInput",
    "Button", "ProgressBar", "SliderBar",
    "DrawingCanvas", "Tilemap",

            "name": "Default",
            "isLooping": False,
            "isPingPong": False,

            "repeatTo": 0,

        self,
        name: str,
        plugin_id: str,

        behaviors: list[dict] | None = None,
        effects: list[dict] | None = None,
        instance_variables: list[dict] | None = None,
        is_global: bool = False,
```

```python
"""
Placeholder PNG image generator for C3 clipboard imageData.

Generates simple colored shapes as base64 PNG data URIs.
"""
from __future__ import annotations

import base64
import io

from PIL import Image, ImageDraw

COLORS = {
    "red": (255, 0, 0, 255),
    "green": (0, 255, 0, 255),
    "blue": (0, 0, 255, 255),
    "yellow": (255, 255, 0, 255),
    "cyan": (0, 255, 255, 255),
    "magenta": (255, 0, 255, 255),
    "white": (255, 255, 255, 255),
    "black": (0, 0, 0, 255),
    "gray": (128, 128, 128, 255),
    "orange": (255, 165, 0, 255),
    "purple": (128, 0, 128, 255),
    "brown": (139, 69, 19, 255),
    "pink": (255, 192, 203, 255),
    "lime": (50, 205, 50, 255),
    "navy": (0, 0, 128, 255),
    "teal": (0, 128, 128, 255),
    "gold": (255, 215, 0, 255),
    "silver": (192, 192, 192, 255),
    "transparent": (0, 0, 0, 0),
}


def parse_color(color_str: str) -> tuple[int, int, int, int]:
    if color_str.lower() in COLORS:
        return COLORS[color_str.lower()]
    if color_str.startswith("#"):
        h = color_str[1:]
        if len(h) == 6:
            return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16), 255)
        if len(h) == 8:
            return (int(h[0:2], 16), int(h[2:4], 16), int(h[4:6], 16), int(h[6:8], 16))
    raise ValueError(f"Invalid color: {color_str}")


def generate_placeholder(
    *,
    width: int = 32,
    height: int = 32,
    color: str = "gray",
    shape: str = "rectangle",
) -> str:
    """Generate a placeholder PNG as a base64 data URI."""
    rgba = parse_color(color)

    if shape == "circle":
        img = Image.new("RGBA", (width, height), (0, 0, 0, 0))
        draw = ImageDraw.Draw(img)
        draw.ellipse([0, 0, width - 1, height - 1], fill=rgba)
    else:
        img = Image.new("RGBA", (width, height), rgba)

    buf = io.BytesIO()
    img.save(buf, format="PNG")
    b64 = base64.b64encode(buf.getvalue()).decode("ascii")
    return f"data:image/png;base64,{b64}"
```

```python
"""
Event sheet generator: Intent IR → C3 events clipboard JSON.

The IR format is a simplified representation:
- variables: list of {name, type, initialValue, comment}
- events: list of {conditions, actions, children?}
- groups: list of {title, description, events}
- functions: list of {name, returnType, parameters, actions}
"""
from __future__ import annotations

from typing import Any

from src.generator.builder import (
    build_action,
    build_block,
    build_comment,
    build_condition,
    build_function_block,
    build_group,
    build_script_action,
    build_variable,
    wrap_clipboard,
)


class EventSheetGenerator:
    def from_ir(self, ir: dict) -> dict:
        """Convert Intent IR to clipboard events JSON."""
        items: list[dict] = []

        # Variables first
        for v in ir.get("variables", []):
            items.append(build_variable(
                v["name"], v.get("type", "number"),
                v.get("initialValue", "0"), v.get("comment", ""),
            ))

        # Top-level events
        for ev in ir.get("events", []):
            items.append(self._build_event(ev))

        # Grouped events
        for g in ir.get("groups", []):
            children = [self._build_event(ev) for ev in g.get("events", [])]
            items.append(build_group(
                g["title"],
                children=children,
                description=g.get("description", ""),
            ))

        # Functions
        for f in ir.get("functions", []):
            actions = [self._build_action_node(a) for a in f.get("actions", [])]
            items.append(build_function_block(
                f["name"],
                return_type=f.get("returnType", "none"),
                parameters=f.get("parameters", []),
                actions=actions,
                description=f.get("description", ""),
                is_async=f.get("isAsync", False),
            ))

        return wrap_clipboard("events", items)

    def _build_event(self, ev: dict) -> dict:
        conditions = [self._build_condition_node(c) for c in ev.get("conditions", [])]
        actions = [self._build_action_node(a) for a in ev.get("actions", [])]
        children = [self._build_event(ch) for ch in ev.get("children", [])] or None
        return build_block(conditions, actions, children=children)

    def _build_condition_node(self, c: dict) -> dict:
        return build_condition(
            c["id"], c["objectClass"],
            parameters=c.get("parameters") or None,
            behavior_type=c.get("behaviorType"),
            inverted=c.get("isInverted", False),
        )

    def _build_action_node(self, a: dict) -> dict[str, Any]:
        # Script action
        if a.get("type") == "script":
            return build_script_action(a["script"], a.get("language", "javascript"))
        # Function call
        if "callFunction" in a:
            node: dict[str, Any] = {"callFunction": a["callFunction"]}
            if "parameters" in a:
                node["parameters"] = a["parameters"]
            return node
        # Regular action
        return build_action(
            a["id"], a["objectClass"],
            parameters=a.get("parameters") or None,
            behavior_type=a.get("behaviorType"),
        )
```

```python
"""
Top-level IR → clipboard JSON renderer.

Dispatches to appropriate generator based on IR type.
"""
from __future__ import annotations

from typing import Any

from src.generator.events import EventSheetGenerator
from src.generator.objects import ObjectTypeGenerator
from src.imagedata.generator import generate_placeholder
from src.validator.structural import StructuralValidator


_validator = StructuralValidator()
_event_gen = EventSheetGenerator()
_object_gen = ObjectTypeGenerator()


def render_ir(
    ir: dict,
    *,
    options: dict | None = None,
) -> dict[str, Any]:
    """Render Intent IR to clipboard JSON with validation and metadata."""
    ir_type = ir.get("type", "")
    opts = options or {}

    try:
        if ir_type == "event_sheet":
            clip = _event_gen.from_ir(ir)
            metadata = _event_metadata(clip)
        elif ir_type == "object_types":
            clip, metadata = _render_objects(ir, opts)
        else:
            return {"success": False, "error": f"Unknown IR type: '{ir_type}'"}
    except Exception as e:
        return {"success": False, "error": str(e)}

    validation = _validator.validate(clip)

    return {
        "success": validation.passed,
        "clipboard_json": clip,
        "validation": {
            "passed": validation.passed,
            "errors": validation.errors,
            "warnings": validation.warnings,
        },
        "metadata": metadata,
    }


def _render_objects(ir: dict, opts: dict) -> tuple[dict, dict]:
    items = []
    for obj_spec in ir.get("objects", []):
        item = _object_gen.build(
            obj_spec["name"],
            obj_spec.get("plugin", "Sprite"),
            behaviors=obj_spec.get("behaviors"),
            effects=obj_spec.get("effects"),
            instance_variables=obj_spec.get("instance_variables"),
            is_global=obj_spec.get("is_global", False),
        )
        items.append(item)

    image_data = None
    if opts.get("include_imagedata", False):
        image_data = [generate_placeholder() for _ in items if "animations" in _]

    clip = _object_gen.build_clipboard(items, image_data=image_data)
    metadata = {
        "objects_created": len(items),
        "behaviors_used": _collect_behaviors(items),
    }
    return clip, metadata


def _event_metadata(clip: dict) -> dict:
    items = clip.get("items", [])
    blocks = sum(1 for i in items if i.get("eventType") == "block")
    variables = sum(1 for i in items if i.get("eventType") == "variable")
    groups = sum(1 for i in items if i.get("eventType") == "group")
    functions = sum(1 for i in items if i.get("eventType") == "function-block")
    return {
        "events_created": blocks,
        "variables_created": variables,
        "groups_created": groups,
        "functions_created": functions,
    }


def _collect_behaviors(items: list[dict]) -> list[str]:
    behaviors = set()
    for item in items:
        for bt in item.get("behaviorTypes", []):
            name = bt.get("name", bt.get("behaviorId", ""))
            if name:
                behaviors.add(name)
    return sorted(behaviors)
```

```python
"""
Error notebook: JSONL-based error case storage.

Append-only log of errors encountered during validation/generation.
Each entry tracks source, error details, status, and fix references.
"""
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path


class ErrorNotebook:
    def __init__(self, base_dir: Path):
        self.cases_dir = base_dir / "cases"
        self.cases_dir.mkdir(parents=True, exist_ok=True)
        self.jsonl_path = self.cases_dir / "errors.jsonl"
        self._counter = self._count_existing()

    def _count_existing(self) -> int:
        if not self.jsonl_path.exists():
            return 0
        return sum(1 for _ in self.jsonl_path.open("r", encoding="utf-8") if _.strip())

    def report(
        self,
        *,
        source: str,
        error_message: str,
        error_type: str,
        input_ir: dict | None = None,
        bad_json: dict | None = None,
    ) -> str:
        self._counter += 1
        entry_id = f"ERR-{datetime.now(timezone.utc).strftime('%Y')}-{self._counter:04d}"
        entry = {
            "id": entry_id,
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "source": source,
            "input_ir": input_ir,
            "bad_json": bad_json,
            "error": {
                "type": error_type,
                "message": error_message,
            },
            "status": "pending",
            "root_cause": None,
            "fix": None,
            "rule_ref": None,
            "test_ref": None,
        }
        with open(self.jsonl_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(entry, ensure_ascii=False) + "\n")
        return entry_id

    def get_pending(self) -> list[dict]:
        return [e for e in self._read_all() if e.get("status") == "pending"]

    def get_stats(self) -> dict:
        entries = self._read_all()
        by_type: dict[str, int] = {}
        by_status: dict[str, int] = {}
        for e in entries:
            t = e.get("error", {}).get("type", "unknown")
            by_type[t] = by_type.get(t, 0) + 1
            s = e.get("status", "unknown")
            by_status[s] = by_status.get(s, 0) + 1
        return {
            "total": len(entries),
            "by_type": by_type,
            "by_status": by_status,
        }

    def _read_all(self) -> list[dict]:
        if not self.jsonl_path.exists():
            return []
        entries = []
        for line in self.jsonl_path.read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if line:
                entries.append(json.loads(line))
        return entries
```

```python
"""
Construct3-Clipboard FastAPI Service

Schema-driven C3 clipboard JSON generation and validation.
Port: 8766
"""
from __future__ import annotations

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from src.config import ERRORS_DIR
from src.errors.notebook import ErrorNotebook
from src.generator.renderer import render_ir
from src.schemas.api import (
    ErrorReportRequest,
    GenerateRequest,
    GenerateResponse,
    HealthResponse,
    ValidateRequest,
    ValidateResponse,
)
from src.validator.structural import StructuralValidator, VALID_CLIPBOARD_TYPES

app = FastAPI(
    title="Construct3-Clipboard",
    description="Schema-driven C3 clipboard JSON generation & validation service",
    version="0.1.0",
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

_validator = StructuralValidator()
_notebook = ErrorNotebook(ERRORS_DIR)


@app.get("/health", response_model=HealthResponse)
async def health():
    return HealthResponse(status="ok")


@app.post("/generate", response_model=GenerateResponse)
async def generate(req: GenerateRequest):
    ir_dict = req.intent_ir.model_dump(exclude_none=True)
    options = req.options or {}
    result = render_ir(ir_dict, options=options)

    if not result.get("success"):
        return GenerateResponse(success=False, error=result.get("error", "Unknown error"))

    return GenerateResponse(
        success=True,
        clipboard_json=result["clipboard_json"],
        validation=result.get("validation"),
        metadata=result.get("metadata"),
    )


@app.post("/validate", response_model=ValidateResponse)
async def validate(req: ValidateRequest):
    result = _validator.validate(req.clipboard_json)
    # Auto-report failures to error notebook
    if not result.passed:
        _notebook.report(
            source="validation_catch",
            error_message="; ".join(result.errors[:3]),
            error_type="structural",
            bad_json=req.clipboard_json,
        )
    return ValidateResponse(
        passed=result.passed,
        errors=result.errors,
        warnings=result.warnings,
    )

    return {
        "valid_types": sorted(VALID_CLIPBOARD_TYPES),

            "Empty parameters should be omitted, not {}",
```
